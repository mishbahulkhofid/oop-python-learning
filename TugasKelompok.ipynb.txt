        "#Tugas Kelompok\n",
        "\n",
        "from abc import ABC, abstractmethod\n",
        "\n",
        "class KamarHotel(ABC):\n",
        "\n",
        "    def __init__(self, nama_kamar, harga_dasar):\n",
        "        self.nama_kamar = nama_kamar\n",
        "        self.__stok = 0\n",
        "        self.__harga_dasar = harga_dasar\n",
        "\n",
        "    def get_stok(self):\n",
        "        return self.__stok\n",
        "\n",
        "    def get_harga_dasar(self):\n",
        "        return self.__harga_dasar\n",
        "\n",
        "    def tambah_stok(self, jumlah):\n",
        "        if jumlah < 0:\n",
        "            print(\n",
        "                f\"Gagal update stok {self.nama_kamar}! \"\n",
        "                f\"Stok tidak boleh negatif ({jumlah}).\"\n",
        "            )\n",
        "        else:\n",
        "            self.__stok += jumlah\n",
        "            print(\n",
        "                f\"Berhasil menambahkan stok {self.nama_kamar}: \"\n",
        "                f\"{jumlah} unit.\"\n",
        "            )\n",
        "\n",
        "    @abstractmethod\n",
        "    def tampilkan_detail(self):\n",
        "        pass\n",
        "\n",
        "    @abstractmethod\n",
        "    def hitung_harga_total(self, jumlah_malam):\n",
        "        pass\n",
        "\n",
        "#Kamar Deluxe\n",
        "class KamarDeluxe(KamarHotel):\n",
        "\n",
        "    def __init__(self, nama_kamar, harga_dasar, fasilitas):\n",
        "        super().__init__(nama_kamar, harga_dasar)\n",
        "        self.fasilitas = fasilitas\n",
        "\n",
        "    def tampilkan_detail(self):\n",
        "        pajak = self.get_harga_dasar() * 0.10\n",
        "\n",
        "        print(\n",
        "            f\"[DELUXE] {self.nama_kamar} | \"\n",
        "            f\"Fasilitas: {self.fasilitas}\"\n",
        "        )\n",
        "        print(\n",
        "            f\"Harga Dasar/Malam: Rp {self.get_harga_dasar():,.0f} | \"\n",
        "            f\"Pajak(10%): Rp {pajak:,.0f}\"\n",
        "        )\n",
        "\n",
        "    def hitung_harga_total(self, jumlah_malam):\n",
        "        harga_per_malam = self.get_harga_dasar()\n",
        "        pajak = harga_per_malam * 0.10\n",
        "\n",
        "        return (harga_per_malam + pajak) * jumlah_malam\n",
        "\n",
        "#Kamar Standard\n",
        "class KamarStandard(KamarHotel):\n",
        "\n",
        "    def __init__(self, nama_kamar, harga_dasar, kapasitas):\n",
        "        super().__init__(nama_kamar, harga_dasar)\n",
        "        self.kapasitas = kapasitas\n",
        "\n",
        "    def tampilkan_detail(self):\n",
        "        pajak = self.get_harga_dasar() * 0.05\n",
        "\n",
        "        print(\n",
        "            f\"[STANDARD] {self.nama_kamar} | \"\n",
        "            f\"Kapasitas: {self.kapasitas}\"\n",
        "        )\n",
        "        print(\n",
        "            f\"Harga Dasar/Malam: Rp {self.get_harga_dasar():,.0f} | \"\n",
        "            f\"Pajak(5%): Rp {pajak:,.0f}\"\n",
        "        )\n",
        "\n",
        "    def hitung_harga_total(self, jumlah_malam):\n",
        "        harga_per_malam = self.get_harga_dasar()\n",
        "        pajak = harga_per_malam * 0.05\n",
        "\n",
        "        return (harga_per_malam + pajak) * jumlah_malam\n",
        "\n",
        "#Proses Pesanan\n",
        "def proses_transaksi(daftar_pesanan):\n",
        "\n",
        "    total_tagihan = 0\n",
        "\n",
        "    print(\"\\n--- STRUK PEMESANAN ---\")\n",
        "\n",
        "    for nomor, pesanan in enumerate(daftar_pesanan, start=1):\n",

        "        kamar = pesanan[\"kamar\"]\n",
        "        jumlah_malam = pesanan[\"malam\"]\n",
        "\n",
        "        kamar.tampilkan_detail()\n",
        "\n",
        "        subtotal = kamar.hitung_harga_total(jumlah_malam)\n",
        "\n",
        "        print(\n",

        "            f\"Subtotal: Rp {subtotal:,.0f}\"\n",
        "        )\n",
        "\n",
        "        print()\n",

        "\n",
        "    print(\"----------------------------------------\")\n",
        "    print(f\"TOTAL TAGIHAN: Rp {total_tagihan:,.0f}\")\n",
        "    print(\"----------------------------------------\")\n",
